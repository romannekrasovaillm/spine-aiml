"""Комиссия по базисным пунктам (AD-2: деньги — через Decimal, ROUND_HALF_UP)."""

from decimal import ROUND_HALF_UP, Decimal

_KOPECK = Decimal("0.01")
_BPS_BASE = Decimal("10000")


def calc_fee(amount: float, bps: float) -> float:
    """Комиссия с суммы `amount` по ставке `bps` базисных пунктов (bps/10000).

    Результат округляется до копеек по правилу ROUND_HALF_UP; промежуточные
    вычисления — на decimal.Decimal (входные float конвертируются через str,
    чтобы не тащить двоичную погрешность в расчёт).
    """
    fee = Decimal(str(amount)) * Decimal(str(bps)) / _BPS_BASE
    return float(fee.quantize(_KOPECK, rounding=ROUND_HALF_UP))

# Тесты агента (tests/test_fee.py)
from bankcalc.fee import calc_fee


def test_fee_common_case():
    # 1000.0 * 50 / 10000 = 5.00
    assert calc_fee(1000.0, 50.0) == 5.0


def test_fee_rounds_down_below_half_kopeck():
    # 1234.56 * 10 / 10000 = 1.23456 -> 1.23
    assert calc_fee(1234.56, 10.0) == 1.23


def test_fee_zero_amount():
    assert calc_fee(0.0, 250.0) == 0.0


def test_fee_half_kopeck_rounds_up():
    # 0.25 * 1000 / 10000 = 0.025 — ровно половина копейки -> вверх (0.03),
    # при ROUND_HALF_EVEN было бы 0.02
    assert calc_fee(0.25, 1000.0) == 0.03


def test_fee_large_amount():
    # 999999999.99 * 12.5 / 10000 = 1249999.9999875 -> 1250000.00
    # (на float-арифметике здесь погрешность в копейках)
    assert calc_fee(999_999_999.99, 12.5) == 1250000.0
